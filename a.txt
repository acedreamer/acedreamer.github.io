1. LEXICAL ANALYZER FOR C (IN C)
================================================================================

Input File: sample.c
--------------------
#include<stdio.h>
void main()
{
int a=1,b=2;
printf("The sum of %d and %d is:%d",a,b,a+b);
}


Running Commands:
-----------------
gcc lexical.c -o lexical
./lexical


Code: lexical.c
---------------
#include<stdio.h>
#include<string.h>
#include<ctype.h>

FILE *fp;
int lineno=0;
char c;
char lexbuf[50],symtab[50][20];
int i=0,x;
char kw[30][20]={"void","int","float","double","short","long","if","else","switch","case","break","return","main","static","goto"};
char delim[]={'(',')','{','}','[',']',';',','};
char oper[]={'+','=','-','*','/','<','>'};

int isdelim(char);
int isoper(char);
int iskw(char[]);
int lookup(char s[]);

int main()
{
    fp=fopen("sample.c","r");
    if(fp==NULL) {
        printf("Error: Cannot open sample.c\n");
        return 1;
    }
    c=getc(fp);
    while(c!=EOF)
    {
        if(c==' '||c=='\t');
        else if(c=='\n')
        {
            lineno++;
        }
        else if((x=isdelim(c))!=-1)
        {
            printf("%c\t\tDelimiter\n",c);
        }
        else if((x=isoper(c))!=-1)
        {
            printf("%c\t\tOperator\n",c);
        }
        else if(isdigit(c))
        {
            int b=0;
            while(isdigit(c))
            {
                lexbuf[b++]=c;
                c=getc(fp);
            }
            ungetc(c,fp);
            lexbuf[b]='\0';
            printf("%s\t\tDigit\n",lexbuf);
        }
        else if(isalpha(c))
        {
            int b=0,k;
            while(isalpha(c)||isdigit(c)||c=='_')
            {
                lexbuf[b++]=c;
                c=getc(fp);
            }
            ungetc(c,fp);
            lexbuf[b]='\0';
            if((!(lookup(lexbuf)))&&(!iskw(lexbuf)))
            {
                strcpy(symtab[i++],lexbuf);
            }
            if((k=iskw(lexbuf))!=0)
            {
                printf("%s\t\t Keyword\n",lexbuf);
            }
            else
            {
                printf("%s\t\tIdentifier\n",lexbuf);
            }
        }
        c=getc(fp);
    }
    fclose(fp);
    printf("\nNumber of lines=%d\n",lineno);
    return 0;
}

int isdelim(char d)
{
    int k;
    for(k=0;k<8;k++)
    {
        if(d==delim[k])
        {
            return k;
        }
    }
    return -1;
}

int isoper(char op)
{
    int k;
    for(k=0;k<7;k++)
    {
        if(op==oper[k])
        {
            return k;
        }
    }
    return -1;
}

int lookup(char s[])
{
    int k;
    for(k=0;k<i;k++)
    {
        if((strcmp(s,symtab[k]))==0)
        {
            return k+1;
        }
    }
    return 0;
}

int iskw(char s[])
{
    int k;
    for(k=0;k<15;k++)
    {
        if(strcmp(s,kw[k])==0)
            return k+1;
    }
    return 0;
}

-------------------------------

#include "y.tab.h"
extern int yylval;
%}

%%
[0-9]+      { yylval = atoi(yytext); return NUMBER; }
[\n]        { return 0; }
[ \t]       ; /* Ignore whitespace */
.           { return yytext[0]; }
%%

int yywrap() {
    return 1;
}


Code: calc.y (YACC Grammar & Evaluator)
---------------------------------------
%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int flag = 0;
int yylex();
void yyerror(const char *s);
%}

%token NUMBER
%left '+' '-'
%left '*' '/' '%'
%left '(' ')'

%%
Calculate : E {
    printf("The calculated value is: %d\n", $$);
}
;

E : E '+' E    { $$ = $1 + $3; }
  | E '-' E    { $$ = $1 - $3; }
  | E '*' E    { $$ = $1 * $3; }
  | E '/' E    { 
        if ($3 == 0) {
            printf("\nRuntime Error: Division by zero!\n");
            flag = 1;
            YYABORT;
        } else {
            $$ = $1 / $3; 
        }
    }
  | E '%' E    { $$ = $1 % $3; }
  | '(' E ')'  { $$ = $2; }
  | NUMBER     { $$ = $1; }
  ;
%%

int main() {
    printf("Enter the arithmetic expression to evaluate:\n");
    yyparse();
    if (flag == 0) {
        printf("Successful evaluation done.\n");
    }
    return 0;
}

void yyerror(const char *s) {
    flag = 1;
    printf("Invalid expression to calculate, unable to parse.\n");
}

3. CONSTANT PROPAGATION
================================================================================

Running Commands:
-----------------
gcc constant_propagation.c -o const_prop
./const_prop


Input Format / Example:
-----------------------
4
= 3 ? a
+ a 5 b
+ b x c
* c 2 d


Code: constant_propagation.c
----------------------------
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

void input();
void constant();
void change(int i, int res);
void output();

struct expression {
    char op[2];
    char op1[5];
    char op2[5];
    char res[5];
    int flag;
} arr[10];

int n;

int main() {
    input();
    constant();
    output();
    return 0;
}

void input() {
    printf("Enter the number of expressions:\n");
    scanf("%d", &n);
    printf("Enter the expressions in format: <op> <op1> <op2> <res>\n");
    printf("(Use '=' with '?' as dummy op2 for assignment, e.g. '= 3 ? a')\n");
    for (int i = 0; i < n; i++) {
        scanf("%s %s %s %s", arr[i].op, arr[i].op1, arr[i].op2, arr[i].res);
        arr[i].flag = 0;
    }
}

void constant() {
    int op1, op2, res;
    char op;
    for (int i = 0; i < n; i++) {
        if ((isdigit(arr[i].op1[0]) && isdigit(arr[i].op2[0])) || 
            (strcmp(arr[i].op, "=") == 0 && isdigit(arr[i].op1[0]))) {
            
            op1 = atoi(arr[i].op1);
            op2 = atoi(arr[i].op2);
            op = arr[i].op[0];
            
            switch (op) {
                case '+': res = op1 + op2; break;
                case '-': res = op1 - op2; break;
                case '*': res = op1 * op2; break;
                case '/': 
                    if (op2 != 0) res = op1 / op2; 
                    else res = 0;
                    break;
                case '=': res = op1; break;
            }
            arr[i].flag = 1;
            change(i, res);
        }
    }
}

void change(int i, int res) {
    char res1[5];
    sprintf(res1, "%d", res);
    for (int j = i + 1; j < n; j++) {
        if (strcmp(arr[i].res, arr[j].op1) == 0) {
            strcpy(arr[j].op1, res1);
        }
        if (strcmp(arr[i].res, arr[j].op2) == 0) {
            strcpy(arr[j].op2, res1);
        }
    }
}

void output() {
    printf("\nOptimized code after Constant Propagation:\n");
    for (int i = 0; i < n; i++) {
        if (arr[i].flag == 0) {
            printf("%s %s %s %s\n", arr[i].op, arr[i].op1, arr[i].op2, arr[i].res);
        }
    }
}

4. FIRST SET OF GRAMMAR
================================================================================

Running Commands:
-----------------
gcc first_of_grammar.c -o first_set
./first_set


Input Format / Example:
-----------------------
8
E=TX
X=+TX
X=$
T=FY
Y=*FY
Y=$
F=(E)
F=i


Code: first_of_grammar.c
------------------------
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

char p[20][20];
char f[20];
int m = 0, n;

void first(char c);
void add_to_result(char ch);

int main() {
    int choice = 1;
    char c;
    
    printf("Enter the number of productions: ");
    scanf("%d", &n);
    
    printf("Enter the productions in (LHS=RHS) format (use '$' for epsilon):\n");
    for (int i = 0; i < n; i++) {
        scanf("%s", p[i]);
    }
    
    do {
        m = 0;
        printf("\nEnter the non-terminal to find FIRST set: ");
        scanf(" %c", &c);
        
        first(c);
        
        printf("FIRST(%c) = { ", c);
        for (int i = 0; i < m; i++) {
            printf("%c", f[i]);
            if (i < m - 1) printf(", ");
        }
        printf(" }\n");
        
        printf("\nDo you want to continue? (1 for Yes / 0 for No): ");
        scanf("%d", &choice);
    } while (choice == 1);
    
    return 0;
}

void add_to_result(char ch) {
    // Avoid duplicate insertions
    for (int i = 0; i < m; i++) {
        if (f[i] == ch) return;
    }
    f[m++] = ch;
}

void first(char c) {
    // If c is terminal or epsilon ($), add directly to result
    if (!isupper(c)) {
        add_to_result(c);
        return;
    }
    
    // c is a non-terminal: search all productions with LHS == c
    for (int i = 0; i < n; i++) {
        if (p[i][0] == c) {
            // If RHS derives epsilon directly
            if (p[i][2] == '$') {
                add_to_result('$');
            } 
            // If RHS begins with terminal
            else if (!isupper(p[i][2])) {
                add_to_result(p[i][2]);
            } 
            // If RHS begins with non-terminal
            else {
                first(p[i][2]);
            }
        }
    }
}

5. INTERMEDIATE CODE GENERATOR (THREE-ADDRESS CODE)
================================================================================

Running Commands:
-----------------
gcc intermediate_code.c -o icg
./icg


Input Format / Example:
-----------------------
w=a*b+c


Code: intermediate_code.c
-------------------------
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

char str[100], left[15], right[15], resultvar[15];
int i = 0, j = 0, tmpch = 'Z'; // Use 'z' for lowercase or 'Z' to match your written output

struct exp {
    int pos;
    char op;
} k[15];

void findopr();
void explore();
void fleft(int x);
void fright(int x);

int main() {
    printf("Enter the exp: ");
    scanf("%s", str);
    
    char *eq_pos = strchr(str, '=');
    if (!eq_pos) {
        printf("invalid exp. '=' missing.\n");
        return 1;
    }
    
    *eq_pos = '\0';
    strcpy(resultvar, str);
    strcpy(str, eq_pos + 1);
    
    printf("\nThe intermediate code:\n");
    findopr();
    explore();
    
    return 0;
}

void findopr() {
    int len = strlen(str);
    for (i = 0; i < len; i++)
        if (str[i] == '*') { k[j].pos = i; k[j++].op = '*'; }
    for (i = 0; i < len; i++)
        if (str[i] == '/') { k[j].pos = i; k[j++].op = '/'; }
    for (i = 0; i < len; i++)
        if (str[i] == '+') { k[j].pos = i; k[j++].op = '+'; }
    for (i = 0; i < len; i++)
        if (str[i] == '-') { k[j].pos = i; k[j++].op = '-'; }
}

void explore() {
    char temp;
    for (i = 0; i < j; i++) {
        fleft(k[i].pos);
        fright(k[i].pos);
        temp = tmpch--;
        str[k[i].pos] = temp;
        printf("%c:=%s%c%s\n", temp, left, k[i].op, right);
    }
    if (j > 0)
        printf("%s:=%c\n", resultvar, tmpch + 1);
    else
        printf("%s:=%s\n", resultvar, str);
}

void fleft(int x) {
    int w = 0;
    x--;
    while (x > 0 && str[x] == ' ')
        x--;
    if (str[x] != ' ') {
        left[w++] = str[x];
        str[x] = ' ';
    }
    left[w] = '\0';
}

void fright(int x) {
    int w = 0;
    x++;
    while (str[x] == ' ')
        x++;
    if (str[x] != '\0' && str[x] != ' ') {
        right[w++] = str[x];
        str[x] = ' ';
    }
    right[w] = '\0';
}

6. SHIFT-REDUCE PARSER
================================================================================

Running Commands:
-----------------
gcc shift_reduce_parser.c -o srp
./srp


Input Format / Example:
-----------------------
id+id*id


Code: shift_reduce_parser.c
---------------------------
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

char input[50], stack[50], action[50];
int k = 0, i = 0, len = 0;

int check() {
    if (k >= 2 && stack[k - 1] == 'd' && stack[k - 2] == 'i') {
        stack[k - 2] = 'E';
        stack[k - 1] = '\0';
        k = k - 1;
        strcpy(action, "REDUCE");
        printf("%s\t\t%s\t\t%s E->id\n", stack, input, action);
        return 1;
    }
    else if (k >= 3 && stack[k - 1] == 'E' && stack[k - 2] == '+' && stack[k - 3] == 'E') {
        stack[k - 3] = 'E';
        stack[k - 2] = '\0';
        k = k - 2;
        strcpy(action, "REDUCE");
        printf("%s\t\t%s\t\t%s E->E+E\n", stack, input, action);
        return 1;
    }
    else if (k >= 3 && stack[k - 1] == 'E' && stack[k - 2] == '*' && stack[k - 3] == 'E') {
        stack[k - 3] = 'E';
        stack[k - 2] = '\0';
        k = k - 2;
        strcpy(action, "REDUCE");
        printf("%s\t\t%s\t\t%s E->E*E\n", stack, input, action);
        return 1;
    }
    else if (k >= 3 && stack[k - 1] == ')' && stack[k - 2] == 'E' && stack[k - 3] == '(') {
        stack[k - 3] = 'E';
        stack[k - 2] = '\0';
        k = k - 2;
        strcpy(action, "REDUCE");
        printf("%s\t\t%s\t\t%s E->(E)\n", stack, input, action);
        return 1;
    }
    return 0;
}

int main() {
    printf("The grammar used to parse is as follows:\n");
    printf("  E -> E + E\n");
    printf("  E -> E * E\n");
    printf("  E -> (E)\n");
    printf("  E -> id\n\n");
    
    printf("Enter the string to parse (e.g., id+id*id):\n");
    scanf("%s", input);
    len = strlen(input);
    
    printf("\nSTACK\t\tINPUT\t\tACTION\n");
    printf("-----\t\t-----\t\t------\n");
    
    while (i < len || !(stack[0] == 'E' && stack[1] == '\0')) {
        if (input[i] == 'i' && input[i + 1] == 'd' && (i + 1) < len) {
            stack[k] = input[i];
            stack[k + 1] = input[i + 1];
            stack[k + 2] = '\0';
            input[i] = ' ';
            input[i + 1] = ' ';
            k += 2;
            i += 2;
            strcpy(action, "SHIFT");
            printf("%s\t\t%s\t\t%s id\n", stack, input, action);
            while (check());
        } 
        else {
            if (!check()) {
                if (i < len) {
                    stack[k] = input[i];
                    char temp = input[i];
                    stack[k + 1] = '\0';
                    input[i] = ' ';
                    k++;
                    i++;
                    strcpy(action, "SHIFT");
                    printf("%s\t\t%s\t\t%s %c\n", stack, input, action, temp);
                    while (check());
                } else {
                    printf("\nREJECT: String cannot be parsed with given grammar.\n");
                    return 0;
                }
            }
        }
    }
    
    if (stack[0] == 'E' && stack[1] == '\0') {
        printf("\nACCEPT: Input string parsed successfully!\n");
    } else {
        printf("\nREJECT: Parsing terminated with error.\n");
    }
    
    return 0;
}
